apply_operator(+, Left, Right, Value) :-\
&#x20;   Value is Left + Right.\
\
apply_operator(-, Left, Right, Value) :-\
&#x20;   Value is Left - Right.\
\
apply_operator(\*, Left, Right, Value) :-\
&#x20;   Value is Left \* Right.\
\
apply_operator(/, Left, Right, Value) :-\
&#x20;   Right =\\= 0,\
&#x20;   0 is Left mod Right,\
&#x20;   Value is Left // Right.
